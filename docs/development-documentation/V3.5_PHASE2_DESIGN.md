## 1. 概述

### 1.1 设计目标

将现有线性工作流升级为 Agent 工作流，实现：

- **图级别循环**: Agent ⇄ Tool 循环在 StateGraph 层面实现
- **对话记忆**: 持久化存储支持多轮对话
- **多工具顺序调用**: 单次请求内多个工具依次执行

### 1.2 当前 vs 目标架构

```
当前架构 (V3.4):
───────────────────────────────────────────────────────────────
START → routeToFunction → queryData → executeCalculation → END
                                          │
                         (工具循环在 generateGuidance 节点内部)

目标架构 (V3.5):
───────────────────────────────────────────────────────────────
                    ┌─────────────────────────────────┐
                    │                                 │
START → loadMemory → agentNode ──────────────────────→ toolNode
                         ↑                               │
                         │                               │
                         └───────── (循环) ──────────────┘
                                      │
                                      ↓ (完成)
                              generateResponse → saveMemory → END
```

### 1.3 关键设计决策

| 决策 | 选择 | 原因 |
|------|------|------|
| 循环位置 | 图级别（条件边） | 符合 LangGraph 设计理念，可观测性更好 |
| 记忆存储 | Supabase PostgreSQL | 复用现有基础设施，支持持久化 |
| 最大循环次数 | 5 次 | 平衡功能与成本/延迟 |
| Session ID 生成 | 后端生成 | 确保唯一性，前端无需依赖 |
| 工具调用 | 顺序执行 | 稳定可靠，V3.6 再考虑并行 |

### 2.2 条件边设计

```typescript
// src/lib/workflows/financial-analysis/graph.ts

/**
 * agentNode 之后的条件边
 * 决定是继续调用工具还是生成最终响应
 */
const afterAgentNode = (state: FinancialAgentGraphState): string => {
  const { agent } = state;

  // 达到最大迭代次数 → 强制生成响应
  if (agent.currentIteration >= agent.maxIterations) {
    console.log(`[Agent] Max iterations (${agent.maxIterations}) reached, generating response`);
    return "generateResponse";
  }

  // Agent 标记完成 → 生成响应
  if (agent.isComplete) {
    console.log("[Agent] Task complete, generating response");
    return "generateResponse";
  }

  // 有待执行的工具调用 → 执行工具
  if (agent.pendingToolCalls.length > 0) {
    console.log(`[Agent] ${agent.pendingToolCalls.length} tool calls pending`);
    return "toolNode";
  }

  // 无工具调用 → 生成响应
  return "generateResponse";
};

/**
 * toolNode 之后的条件边
 * 始终返回 agentNode 让 Agent 决定下一步
 */
const afterToolNode = (_state: FinancialAgentGraphState): string => {
  return "agentNode";
};
```

### 2.3 节点职责

| 节点 | 职责 | 输入 | 输出 |
|------|------|------|------|
| `loadMemory` | 从数据库加载历史消息 | sessionId | agent.messages |
| `agentNode` | LLM 思考，决定调用工具或完成 | messages + question | tool_calls 或 response |
| `toolNode` | 执行待处理的工具调用 | pendingToolCalls | toolResults (追加到 messages) |
| `generateResponse` | 生成最终用户响应 | agent state | output |
| `saveMemory` | 保存对话到数据库 | sessionId + messages | - |

### 2.4 Agent Node 实现

```typescript
// src/lib/workflows/financial-analysis/nodes.ts

import OpenAI from "openai";
import { toolRegistry } from "@/lib/tools";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

/**
 * Agent Node: LLM 思考并决定下一步行动
 */
export const agentNode = async (
  state: FinancialAgentGraphState
): Promise<Partial<FinancialAgentGraphState>> => {
  const { input, agent } = state;

  // 构建消息列表（系统提示 + 历史消息 + 当前问题）
  const systemPrompt = buildAgentSystemPrompt();
  const messages: OpenAI.ChatCompletionMessageParam[] = [
    { role: "system", content: systemPrompt },
    ...agent.messages.map(m => ({
      role: m.role as "user" | "assistant" | "tool",
      content: m.content,
      ...(m.tool_calls && { tool_calls: m.tool_calls }),
      ...(m.tool_call_id && { tool_call_id: m.tool_call_id }),
    })),
  ];

  // 如果是首次迭代，添加当前问题
  if (agent.currentIteration === 0) {
    messages.push({ role: "user", content: input.question });
  }

  // 获取工具定义
  const tools = toolRegistry.toOpenAIFunctionDefs({
    category: "financial",
    detailed: true
  });

  // 调用 LLM
  const completion = await openai.chat.completions.create({
    model: "gpt-5-mini",
    messages,
    tools: tools.length > 0 ? tools : undefined,
    tool_choice: tools.length > 0 ? "auto" : undefined,
    max_completion_tokens: 2000,
  });

  const responseMessage = completion.choices[0].message;

  // 更新 Agent 状态
  const newMessages = [...agent.messages];

  // 如果是首次迭代，记录用户消息
  if (agent.currentIteration === 0) {
    newMessages.push({
      role: "user",
      content: input.question,
      created_at: new Date().toISOString(),
    });
  }

  // 记录 Assistant 消息
  newMessages.push({
    role: "assistant",
    content: responseMessage.content || "",
    tool_calls: responseMessage.tool_calls?.map(tc => ({
      id: tc.id,
      type: tc.type,
      function: {
        name: tc.function.name,
        arguments: tc.function.arguments,
      },
    })),
    created_at: new Date().toISOString(),
  });

  // 提取待执行的工具调用
  const pendingToolCalls = responseMessage.tool_calls?.map(tc => ({
    id: tc.id,
    name: tc.function.name,
    arguments: JSON.parse(tc.function.arguments),
  })) || [];

  // 判断是否完成
  const isComplete = pendingToolCalls.length === 0 && responseMessage.content !== null;

  return {
    agent: {
      ...agent,
      messages: newMessages,
      currentIteration: agent.currentIteration + 1,
      isComplete,
      pendingToolCalls,
    },
    meta: { ...state.meta, currentStep: "agent_node" },
  };
};
```

### 2.5 Tool Node 实现

```typescript
/**
 * Tool Node: 执行待处理的工具调用
 */
export const toolNode = async (
  state: FinancialAgentGraphState
): Promise<Partial<FinancialAgentGraphState>> => {
  const { agent, input } = state;
  const newMessages = [...agent.messages];
  const toolResults: ToolResult[] = [];

  // 顺序执行每个工具调用
  for (const toolCall of agent.pendingToolCalls) {
    console.log(`[ToolNode] Executing: ${toolCall.name}`);

    try {
      const result = await toolRegistry.execute(toolCall.name, {
        ...toolCall.arguments,
        // 注入上下文（年份、文档ID等）
        year: toolCall.arguments.year || input.year,
        documentIds: input.documentIds,
      });

      const toolMessage = {
        role: "tool" as const,
        tool_call_id: toolCall.id,
        content: JSON.stringify(result.success ? result.data : { error: result.error }),
        created_at: new Date().toISOString(),
      };

      newMessages.push(toolMessage);
      toolResults.push({
        toolCallId: toolCall.id,
        toolName: toolCall.name,
        success: result.success,
        data: result.data,
        error: result.error,
      });

    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : "Unknown error";
      newMessages.push({
        role: "tool" as const,
        tool_call_id: toolCall.id,
        content: JSON.stringify({ error: errorMessage }),
        created_at: new Date().toISOString(),
      });
      toolResults.push({
        toolCallId: toolCall.id,
        toolName: toolCall.name,
        success: false,
        error: errorMessage,
      });
    }
  }

  return {
    agent: {
      ...agent,
      messages: newMessages,
      pendingToolCalls: [], // 清空待执行列表
    },
    toolResults: [...(state.toolResults || []), ...toolResults],
    meta: { ...state.meta, currentStep: "tool_node" },
  };
};
```

```sql
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_session_timestamp
BEFORE UPDATE ON conversation_sessions
FOR EACH ROW
EXECUTE FUNCTION update_session_timestamp();

-- =============================================================================
-- 对话消息表
-- =============================================================================
CREATE TABLE conversation_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID NOT NULL REFERENCES conversation_sessions(id) ON DELETE CASCADE,

  -- 消息角色
  role TEXT NOT NULL CHECK (role IN ('user', 'assistant', 'tool', 'system')),

  -- 消息内容
  content TEXT NOT NULL,

  -- 工具调用（仅 assistant 角色）
  tool_calls JSONB,

  -- 工具调用 ID（仅 tool 角色）
  tool_call_id TEXT,

  -- 元数据
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  sequence_number INTEGER NOT NULL,

  -- 约束
  CONSTRAINT valid_tool_calls CHECK (
    role != 'assistant' OR tool_calls IS NULL OR jsonb_typeof(tool_calls) = 'array'
  ),
  CONSTRAINT valid_tool_role CHECK (
    role != 'tool' OR tool_call_id IS NOT NULL
  ),
  CONSTRAINT unique_sequence UNIQUE (session_id, sequence_number)
);

-- 索引
CREATE INDEX idx_messages_session ON conversation_messages(session_id);
CREATE INDEX idx_messages_session_sequence ON conversation_messages(session_id, sequence_number);
```

### 3.2 Memory Service

```typescript
// src/lib/memory/conversation-memory.ts

import { createServerSupabaseClient } from "../supabase";

export interface ConversationMessage {
  id?: string;
  role: "user" | "assistant" | "tool" | "system";
  content: string;
  tool_calls?: ToolCall[];
  tool_call_id?: string;
  created_at?: string;
  sequence_number?: number;
}

export interface ConversationSession {
  id: string;
  workflowType: string;
  messages: ConversationMessage[];
  metadata: Record<string, unknown>;
  createdAt: string;
  updatedAt: string;
}

/**
 * 对话记忆服务
 */
export class ConversationMemory {
  private supabase = createServerSupabaseClient();

  /**
   * 创建新会话
   */
  async createSession(
    workflowType: string,
    metadata: Record<string, unknown> = {}
  ): Promise<string> {
    const { data, error } = await this.supabase
      .from("conversation_sessions")
      .insert({
        workflow_type: workflowType,
        metadata,
      })
      .select("id")
      .single();

    if (error) throw new Error(`Failed to create session: ${error.message}`);
    return data.id;
  }

  /**
   * 加载会话历史
   * @param sessionId 会话 ID
   * @param limit 最多加载的消息数（默认 10）
   */
  async loadSession(
    sessionId: string,
    limit: number = 10
  ): Promise<ConversationSession | null> {
    // 获取会话信息
    const { data: session, error: sessionError } = await this.supabase
      .from("conversation_sessions")
      .select("*")
      .eq("id", sessionId)
      .single();

    if (sessionError || !session) return null;

    // 获取最近的消息
    const { data: messages, error: messagesError } = await this.supabase
      .from("conversation_messages")
      .select("*")
      .eq("session_id", sessionId)
      .order("sequence_number", { ascending: false })
      .limit(limit);

    if (messagesError) {
      console.error("Failed to load messages:", messagesError);
      return null;
    }

    // 按顺序排列（最早的在前）
    const orderedMessages = (messages || []).reverse();

    return {
      id: session.id,
      workflowType: session.workflow_type,
      messages: orderedMessages.map(m => ({
        id: m.id,
        role: m.role,
        content: m.content,
        tool_calls: m.tool_calls,
        tool_call_id: m.tool_call_id,
        created_at: m.created_at,
        sequence_number: m.sequence_number,
      })),
      metadata: session.metadata || {},
      createdAt: session.created_at,
      updatedAt: session.updated_at,
    };
  }

  /**
   * 保存消息到会话
   */
  async saveMessages(
    sessionId: string,
    messages: ConversationMessage[]
  ): Promise<void> {
    // 获取当前最大 sequence_number
    const { data: maxSeq } = await this.supabase
      .from("conversation_messages")
      .select("sequence_number")
      .eq("session_id", sessionId)
      .order("sequence_number", { ascending: false })
      .limit(1)
      .single();

    let nextSequence = (maxSeq?.sequence_number || 0) + 1;

    // 插入新消息
    const messagesToInsert = messages.map(m => ({
      session_id: sessionId,
      role: m.role,
      content: m.content,
      tool_calls: m.tool_calls || null,
      tool_call_id: m.tool_call_id || null,
      sequence_number: nextSequence++,
    }));

    const { error } = await this.supabase
      .from("conversation_messages")
      .insert(messagesToInsert);

    if (error) throw new Error(`Failed to save messages: ${error.message}`);

    // 更新会话时间戳
    await this.supabase
      .from("conversation_sessions")
      .update({ updated_at: new Date().toISOString() })
      .eq("id", sessionId);
  }

  /**
   * 获取或创建会话
   */
  async getOrCreateSession(
    sessionId: string | undefined,
    workflowType: string
  ): Promise<{ sessionId: string; messages: ConversationMessage[]; isNew: boolean }> {
    if (sessionId) {
      const session = await this.loadSession(sessionId);
      if (session) {
        return {
          sessionId: session.id,
          messages: session.messages,
          isNew: false,
        };
      }
    }

    // 创建新会话
    const newSessionId = await this.createSession(workflowType);
    return {
      sessionId: newSessionId,
      messages: [],
      isNew: true,
    };
  }
}

// 单例导出
export const conversationMemory = new ConversationMemory();
```

### 3.3 Memory Nodes

```typescript
// src/lib/workflows/financial-analysis/nodes.ts

import { conversationMemory } from "@/lib/memory/conversation-memory";

/**
 * Load Memory Node: 从数据库加载历史消息
 */
export const loadMemoryNode = async (
  state: FinancialAgentGraphState
): Promise<Partial<FinancialAgentGraphState>> => {
  const { sessionId } = state.input;

  const { sessionId: resolvedSessionId, messages, isNew } =
    await conversationMemory.getOrCreateSession(sessionId, "financial-analysis");

  console.log(`[LoadMemory] Session: ${resolvedSessionId}, isNew: ${isNew}, messages: ${messages.length}`);

  return {
    agent: {
      ...state.agent,
      messages,
      sessionId: resolvedSessionId,
    },
    meta: { ...state.meta, currentStep: "load_memory" },
  };
};

/**
 * Save Memory Node: 保存对话到数据库
 */
export const saveMemoryNode = async (
  state: FinancialAgentGraphState
): Promise<Partial<FinancialAgentGraphState>> => {
  const { agent } = state;

  if (!agent.sessionId) {
    console.warn("[SaveMemory] No sessionId, skipping save");
    return { meta: { ...state.meta, currentStep: "save_memory" } };
  }

  // 只保存本次会话新增的消息（排除已有的历史消息）
  const newMessages = agent.messages.filter(m => !m.id);

  if (newMessages.length > 0) {
    await conversationMemory.saveMessages(agent.sessionId, newMessages);
    console.log(`[SaveMemory] Saved ${newMessages.length} new messages`);
  }

  return {
    meta: { ...state.meta, currentStep: "save_memory", endTime: Date.now() },
  };
};
```

```typescript
// src/lib/workflows/financial-analysis/state.ts

import { Annotation } from "@langchain/langgraph";
import type { WorkflowMeta } from "@/lib/types/workflow-states";

/**
 * Agent 消息类型
 */
export interface AgentMessage {
  id?: string;
  role: "user" | "assistant" | "tool" | "system";
  content: string;
  tool_calls?: Array<{
    id: string;
    type: string;
    function: { name: string; arguments: string };
  }>;
  tool_call_id?: string;
  created_at?: string;
}

/**
 * 工具调用类型
 */
export interface ToolCall {
  id: string;
  name: string;
  arguments: Record<string, unknown>;
}

/**
 * 工具执行结果
 */
export interface ToolResult {
  toolCallId: string;
  toolName: string;
  success: boolean;
  data?: unknown;
  error?: string;
}

/**
 * Financial Analysis Agent State Annotation
 */
export const FinancialAgentAnnotation = Annotation.Root({
  // 输入
  input: Annotation<{
    question: string;
    sessionId?: string;
    year?: string;
    period?: string;
    documentIds?: string[];
  }>({
    reducer: (prev, next) => next ?? prev,
    default: () => ({ question: "" }),
  }),

  // Agent 状态
  agent: Annotation<{
    sessionId?: string;
    messages: AgentMessage[];
    currentIteration: number;
    maxIterations: number;
    isComplete: boolean;
    pendingToolCalls: ToolCall[];
  }>({
    reducer: (prev, next) => ({ ...prev, ...next }),
    default: () => ({
      messages: [],
      currentIteration: 0,
      maxIterations: 5,
      isComplete: false,
      pendingToolCalls: [],
    }),
  }),

  // 工具执行结果（累积）
  toolResults: Annotation<ToolResult[]>({
    reducer: (prev, next) => [...(prev ?? []), ...(next ?? [])],
    default: () => [],
  }),

  // 最终输出
  output: Annotation<{
    type: "success" | "partial" | "error" | "no_data";
    answer: string;
    sessionId: string;
    calculations?: Array<{
      name: string;
      result: { value: number; formatted: string; unit: string };
      formula: string;
    }>;
    confidence: number;
    sourceDocuments: Array<{ id: string; name: string }>;
    iterationCount: number;
  }>({
    reducer: (prev, next) => next ?? prev,
    default: () => ({
      type: "error",
      answer: "",
      sessionId: "",
      confidence: 0,
      sourceDocuments: [],
      iterationCount: 0,
    }),
  }),

  // 元数据
  meta: Annotation<WorkflowMeta>({
    reducer: (prev, next) => ({
      ...prev,
      ...next,
      errors: [...(prev?.errors ?? []), ...(next?.errors ?? [])],
    }),
    default: () => ({
      startTime: Date.now(),
      currentStep: "init",
      errors: [],
    }),
  }),
});

export type FinancialAgentGraphState = typeof FinancialAgentAnnotation.State;
```

```typescript
    default: () => ({ question: "" }),
  }),

  // Agent 状态（与 Financial Analysis 相同结构）
  agent: Annotation<{
    sessionId?: string;
    messages: AgentMessage[];
    currentIteration: number;
    maxIterations: number;
    isComplete: boolean;
    pendingToolCalls: ToolCall[];
  }>({
    reducer: (prev, next) => ({ ...prev, ...next }),
    default: () => ({
      messages: [],
      currentIteration: 0,
      maxIterations: 5,
      isComplete: false,
      pendingToolCalls: [],
    }),
  }),

  // 搜索结果（保留现有结构）
  search: Annotation<{
    keywords: string[];
    combinedResults: Array<{
      regulation: string;
      article: string;
      content: string;
      relevance: number;
    }>;
    businessContext?: string;
  }>({
    reducer: (prev, next) => ({ ...prev, ...next }),
    default: () => ({ keywords: [], combinedResults: [] }),
  }),

  // 工具执行结果
  toolResults: Annotation<ToolResult[]>({
    reducer: (prev, next) => [...(prev ?? []), ...(next ?? [])],
    default: () => [],
  }),

  output: Annotation<{
    type: "success" | "not_found" | "error";
    guidance: string;
    sessionId: string;
    citations: Array<{ regulation: string; article: string; content: string }>;
    calculations?: Array<{ name: string; result: unknown }>;
    confidence: number;
    iterationCount: number;
  }>({
    reducer: (prev, next) => next ?? prev,
    default: () => ({
      type: "error",
      guidance: "",
      sessionId: "",
      citations: [],
      confidence: 0,
      iterationCount: 0,
    }),
  }),

  meta: Annotation<WorkflowMeta>({
    reducer: (prev, next) => ({
      ...prev,
      ...next,
      errors: [...(prev?.errors ?? []), ...(next?.errors ?? [])],
    }),
    default: () => ({
      startTime: Date.now(),
      currentStep: "init",
      errors: [],
    }),
  }),
});
```

### 5.1 Financial Analysis Agent Prompt

```typescript
// src/lib/prompts/financial-agent-system.txt

你是一个专业的财务分析 Agent，专门帮助用户分析企业财务数据、计算财务指标、提供财务洞察。

## 你的能力

1. **查询财务数据**: 使用 `query_financial_data` 工具获取资产负债表、利润表、现金流量表的数据
2. **计算财务指标**: 使用 `calculate_financial_metric` 工具计算各类财务比率
3. **对比分析**: 利用对话历史进行跨期间对比（如"与去年相比"）

## 可用工具

### query_financial_data
查询结构化财务数据。
- `metric_names`: 指标名称列表（如 ["资产总计", "负债总计"]）
- `year`: 年份（可选，如 "2024"）
- `period`: 期间（可选，如 "annual", "Q1"）

### calculate_financial_metric
使用预设公式计算财务指标。
- `metric_name`: 指标英文名（如 "debt_to_asset_ratio"）
- `inputs`: 计算所需的数值（如 {"负债总计": 5629279672, "资产总计": 14786316939}）

## 决策规则

1. **先查后算**: 总是先使用 query_financial_data 获取数据，再使用 calculate_financial_metric 计算
2. **理解上下文**:
   - 如果用户说"去年"或"上期"，从对话历史中获取之前提到的年份/期间
   - 如果用户说"与...相比"，需要查询两个期间的数据
3. **多指标处理**: 如果用户一次询问多个指标，依次处理每个指标
4. **数据缺失处理**: 如果某些数据不存在，说明哪些数据缺失，并基于已有数据给出部分分析

## 响应格式

- 使用中文回答
- 清晰列出计算过程和结果
- 解释指标的财务含义
- 如果进行对比分析，说明变化趋势和可能原因

## 终止条件

当满足以下任一条件时，停止调用工具并生成最终回答：
- 用户问题已完整解答
- 所有必要的计算已完成
- 遇到无法获取的关键数据（告知用户哪些数据缺失）
- 达到工具调用次数上限

  1. 先查询 负债总计 和 资产总计
  2. 再用公式计算

用户: 与去年相比如何？
你:（思考）用户想对比今年和去年的资产负债率
  从历史对话知道今年是 2024 年
  1. 查询 2023 年的 负债总计 和 资产总计
```

### 6.2 API Route 更新

```typescript
// src/app/api/financial-analysis/route.ts

import { NextRequest, NextResponse } from "next/server";
import { runFinancialAgentWorkflow } from "@/lib/workflows/financial-analysis";
import { validateRequest, FinancialAnalysisRequestSchema } from "@/lib/validation/api-schemas";
import { generateRequestId, createErrorResponse } from "@/lib/security/error-handler";

export async function POST(request: NextRequest) {
  const requestId = request.headers.get("x-request-id") || generateRequestId();

  try {
    const body = await request.json();

    const validation = validateRequest(FinancialAnalysisRequestSchema, body);
    if (!validation.success) {
      return NextResponse.json(
        { error: validation.error, requestId },
        { status: 400, headers: { "X-Request-ID": requestId } }
      );
    }

    const { question, sessionId, year, period, documentIds } = validation.data;

    // 使用 Agent 工作流
    const state = await runFinancialAgentWorkflow({
      question,
      sessionId,  // 传入 sessionId（可能为 undefined）
      year,
      period,
      documentIds,
    });

    // 返回包含 sessionId 的响应
    return NextResponse.json({
      ...state.output,
      sessionId: state.agent.sessionId,  // 确保返回 sessionId
    });

  } catch (error: unknown) {
    console.error("Financial Analysis API error:", error);
    return createErrorResponse(error, requestId, 500);
  }
}
```

## 附录 A: 完整 Agent Graph 示例

```typescript
// src/lib/workflows/financial-analysis/graph.ts

import { StateGraph, START, END } from "@langchain/langgraph";
import { FinancialAgentAnnotation, type FinancialAgentGraphState } from "./state";
import * as nodes from "./nodes";
import { withLogging } from "../shared/graph-utils";

const GRAPH_NAME = "FinancialAgent";

// --- Conditional Edges ---

const afterAgentNode = (state: FinancialAgentGraphState): string => {
  const { agent } = state;

  if (agent.currentIteration >= agent.maxIterations) {
    return "generateResponse";
  }
  if (agent.isComplete) {
    return "generateResponse";
  }
  if (agent.pendingToolCalls.length > 0) {
    return "toolNode";
  }
  return "generateResponse";
};

const afterToolNode = (): string => "agentNode";

// --- Graph Definition ---

const workflow = new StateGraph(FinancialAgentAnnotation)
  // Nodes
  .addNode("loadMemory", withLogging(GRAPH_NAME, "loadMemory", nodes.loadMemoryNode))
  .addNode("agentNode", withLogging(GRAPH_NAME, "agentNode", nodes.agentNode))
  .addNode("toolNode", withLogging(GRAPH_NAME, "toolNode", nodes.toolNode))
  .addNode("generateResponse", withLogging(GRAPH_NAME, "generateResponse", nodes.generateResponseNode))
  .addNode("saveMemory", withLogging(GRAPH_NAME, "saveMemory", nodes.saveMemoryNode))

  // Edges
  .addEdge(START, "loadMemory")
  .addEdge("loadMemory", "agentNode")
  .addConditionalEdges("agentNode", afterAgentNode, {
    toolNode: "toolNode",
    generateResponse: "generateResponse",
  })
  .addConditionalEdges("toolNode", afterToolNode, {
    agentNode: "agentNode",
  })
  .addEdge("generateResponse", "saveMemory")
  .addEdge("saveMemory", END);

// Compile
export const financialAgentGraph = workflow.compile();

// Runner
export async function runFinancialAgentWorkflow(
  input: FinancialAgentGraphState["input"]
): Promise<FinancialAgentGraphState> {
  const result = await financialAgentGraph.invoke({ input });
  return result as FinancialAgentGraphState;
}
```

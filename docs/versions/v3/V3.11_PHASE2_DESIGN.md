## 2. 架构决策

### 2.1 类型文件分离策略

**决策**: 将自动生成类型与自定义类型分离

**结构**:
```
src/lib/database/
├── database.types.ts   # 自动生成 (DO NOT EDIT)
├── custom-types.ts     # 手动维护的扩展
├── helpers.ts          # 类型安全的查询助手
└── types.ts            # 统一导出入口
```

**理由**:
- 防止 `npx supabase gen types` 覆盖自定义类型
- 清晰的关注点分离
- 允许独立更新任一层

### 2.2 Supabase 查询类型策略

**决策**: 使用类型注解和显式字段选择，而非创建复杂的类型包装器

**方法**:
1. 将 `.select('*')` 替换为显式字段列表
2. 对查询结果使用类型注解
3. 为 RPC 函数创建类型安全的包装器

**理由**:
- 最小代码变更
- 保持 Supabase 客户端的标准用法
- 仅在 RPC 函数（类型推断失效处）添加包装器

### 2.3 LangGraph 状态类型策略

**决策**: 创建安全的状态访问器函数，而非重构整个状态系统

**方法**:
1. 创建 `extractStateSummary()` 函数，使用类型守卫安全访问可选字段
2. 创建 `convertToolCallsToDb()` 函数处理格式转换
3. 为 `graph.invoke()` 结果使用正确的类型断言

**理由**:
- 避免大规模 LangGraph 状态重构
- 类型守卫在边界提供运行时安全
- 保持现有工作流逻辑不变

### 2.4 可接受的 `as any` 用法

**决策**: 以下情况保留 `as any` 并添加 REASON 注释

| 场景 | 数量 | 理由 |
|------|------|------|
| XLSX `sheet_to_json()` | 7 | 外部库返回 `unknown[]`，无类型接口 |
| Zod `_def` 访问 | 2 | 必须访问内部 API 进行 schema 内省 |
| JSONB `tool_calls` | 1 | 数据库返回泛型 Json 类型 |
| **总计** | **10** | |

## 5. REQ-002 LangGraph 状态类型设计

### 5.1 状态访问器

**`src/lib/workflows/shared/state-accessors.ts`**:
```typescript
/**
 * 工作流状态中可能存在的可选字段类型
 */
interface RoutingState {
  selectedFunction?: string;
  confidence?: number;
  functionArgs?: { _reasoning?: string };
}

interface DataQueryState {
  missingMetrics?: unknown[];
  error?: string;
}

interface ConfidenceState {
  score?: number;
  passed?: boolean;
}

interface OutputState {
  type?: string;
}

/**
 * 从任意工作流状态中安全提取日志摘要
 * 使用类型守卫确保安全的属性访问
 */
export function extractStateSummary(state: Record<string, unknown>): Record<string, unknown> {
  const summary: Record<string, unknown> = {};

  // 安全提取 routing 信息
  if (state.routing && typeof state.routing === 'object') {
    const routing = state.routing as RoutingState;
    summary.routing = {
      selectedFunction: routing.selectedFunction,
      confidence: routing.confidence,
      reasoning: routing.functionArgs?._reasoning,
    };
  }

  // 安全提取 dataQuery 信息
  if (state.dataQuery && typeof state.dataQuery === 'object') {
    const dataQuery = state.dataQuery as DataQueryState;
    summary.dataQuery = {
      missingCount: Array.isArray(dataQuery.missingMetrics) ? dataQuery.missingMetrics.length : 0,
      hasError: !!dataQuery.error,
    };
  }

  // 安全提取 confidence 信息
  if (state.confidence && typeof state.confidence === 'object') {
    const confidence = state.confidence as ConfidenceState;
    summary.confidence = {
      score: confidence.score,
      passed: confidence.passed,
    };
  }

  // 安全提取 output 信息
  if (state.output && typeof state.output === 'object') {
    const output = state.output as OutputState;
    summary.outputType = output.type;
  }

  return summary;
}
```

## 6. REQ-003 对话记忆类型设计

### 6.1 消息类型定义

**`src/lib/memory/types.ts`**:
```typescript
import type { MessageRole } from '@/lib/database/custom-types';

/**
 * 数据库兼容的工具调用结构
 */
export interface DbToolCall {
  id: string;
  type: 'function';
  function: {
    name: string;
    arguments: string; // JSON 字符串
  };
}

/**
 * LangChain 格式的工具调用
 */
interface LangChainToolCall {
  id?: string;
  name: string;
  args: Record<string, unknown>;
}

/**
 * 将 LangChain 工具调用格式转换为数据库格式
 */
export function convertToolCallsToDb(toolCalls: LangChainToolCall[]): DbToolCall[] {
  return toolCalls.map(tc => ({
    id: tc.id || '',
    type: 'function' as const,
    function: {
      name: tc.name,
      arguments: JSON.stringify(tc.args),
    },
  }));
}

/**
 * 验证消息角色是否有效
 */
export function isValidMessageRole(role: unknown): role is MessageRole {
  return role === 'user' || role === 'assistant' || role === 'tool' || role === 'system';
}
```
